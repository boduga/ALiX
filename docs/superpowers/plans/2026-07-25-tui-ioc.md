Create `src/tui/io.ts`:
```ts
export interface IInput {
  onData(cb: (buf: Buffer) => void): () => void; // returns unsubscribe
}

export interface IOutput {
  write(text: string): void;
}

export class StdioInput implements IInput {
  onData(cb: (buf: Buffer) => void): () => void {
    process.stdin.on('data', cb);
    return () => { process.stdin.off('data', cb); };
  }
}

export class StdioOutput implements IOutput {
  write(text: string): void { process.stdout.write(text); }
}

export class MockInput implements IInput {
  private cbs: Set<(buf: Buffer) => void> = new Set();
  onData(cb: (buf: Buffer) => void): () => void {
    this.cbs.add(cb);
    return () => this.cbs.delete(cb);
  }
  emit(buf: Buffer): void { for (const cb of this.cbs) cb(buf); }
}

export class MockOutput implements IOutput {
  writes: string[] = [];
  write(text: string): void { this.writes.push(text); }
}
```

Modify TuiApp constructor to accept `input` and `output`:

```ts
export interface TuiAppOptions {
  // ...existing fields...
  input?: IInput;    // default: new StdioInput()
  output?: IOutput;  // default: new StdioOutput()
}
```

Update `start()` to use `this.input.onData(...)` instead of `process.stdin.on(...)`.
Update `cleanupSync()` and all `stdout.write` calls to use `this.output.write(...)`.

Write 3 tests:
