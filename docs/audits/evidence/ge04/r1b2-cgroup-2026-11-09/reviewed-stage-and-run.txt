#!/usr/bin/env bash
# Review artifact. Not installed as a privileged helper or sudo target.
# Execute only after independent source/root-scope review, on the designated VM.
set -euo pipefail
qual_review_complete=1
if [[ "$qual_review_complete" != 1 ]]; then
  echo 'Source/root-scope review completion gate is still pending.' >&2
  exit 64
fi
qual_repo=/tmp/alix-ge04-r1b1-mandatory
qual_c_sha=5362bdb0b08586f51151f0b53519a48ababb557f9c31758283f3aab2ec06d757
qual_py_sha=5f65166afc7a66c8d56370b40df32a55f16c82b4adb9fddfaa4eab05985d4174
qual_target=ubuntu@10.1.1.13
qual_remote=/home/ubuntu/alix-ge04-workspaces/90c0042f-cgroup-01
qual_artifacts="$qual_repo/docs/audits/evidence/ge04/r1b2-cgroup-2026-11-09"
qual_ssh=(ssh -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10)
qual_scp=(scp -o BatchMode=yes -o StrictHostKeyChecking=yes -o ConnectTimeout=10)
[[ $(sha256sum "$qual_repo/native/ge04/cgroup-worker-qualification.c" | cut -d' ' -f1) == "$qual_c_sha" ]]
[[ $(sha256sum "$qual_repo/tests/agents/test_ge04_cgroup_worker.py" | cut -d' ' -f1) == "$qual_py_sha" ]]
mkdir -p "$qual_artifacts"
"${qual_ssh[@]}" "$qual_target" 'python3 -I -S -' > "$qual_artifacts/host-before.txt" <<'PYHOST'
import datetime,json,os,pathlib,stat,subprocess
fields=['ActiveState','SubState','InvocationID','MainPID','ExecMainStartTimestampMonotonic']
witness=dict(line.split('=',1) for line in subprocess.check_output(['/usr/bin/systemctl','show','ge04-r2q-witness.service',*[f'--property={f}' for f in fields]],text=True).splitlines())
expected={'ActiveState':'active','SubState':'running','InvocationID':'76b16df891ea439fb850303c99fdf0be','MainPID':'981','ExecMainStartTimestampMonotonic':'20927201'}
unit=subprocess.check_output(['/usr/bin/systemctl','show','ge04-r1b2-qualification.service','--property=LoadState','--property=ActiveState'],text=True)
rows=subprocess.check_output(['/usr/bin/ps','-eo','uid=,gid='],text=True).splitlines()
count=sum(1 for row in rows if 20002 in map(int,row.split()))
paths=['/run/ge04-cgroup-code','/run/ge04-cgroup-qualification','/sys/fs/cgroup/system.slice/ge04-r1b2-qualification.service']
exists={p:os.path.lexists(p) for p in paths}
lookups={k:subprocess.run(['/usr/bin/getent',k,'20002'],capture_output=True,timeout=3).returncode for k in ['passwd','group']}
# Bootstrap uses only these fixed root-owned executables, never a repo root script.
tools=['sudo','env','gnumkdir','gnuchmod','gnurm','gnurmdir','tee','sha256sum','cc','python3','systemd-run','systemctl','gnusleep']
resolved={}
for name in tools:
    original=pathlib.Path('/usr/bin')/name
    actual=original.resolve(strict=True)
    for path in [original,*original.parents,actual,*actual.parents]:
        meta=path.lstat()
        assert meta.st_uid==0 and (stat.S_ISLNK(meta.st_mode) or not meta.st_mode&0o022),str(path)
    assert actual.is_file() and os.access(actual,os.X_OK),str(actual)
    resolved[name]=str(actual)
out={'utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'witness':witness,'unit':unit,'synthetic_process_count':count,'fixture_paths_exist':exists,'identity_lookup_exit':lookups,'resolved_tools':resolved}
print(json.dumps(out,indent=2))
assert witness==expected and count==0 and not any(exists.values()) and all(v==2 for v in lookups.values()) and 'LoadState=not-found\n' in unit
PYHOST
"${qual_ssh[@]}" "$qual_target" "test ! -e '$qual_remote' && mkdir -m 700 '$qual_remote'"
"${qual_scp[@]}" "$qual_repo/native/ge04/cgroup-worker-qualification.c" "$qual_target:$qual_remote/cgroup-worker-qualification.c"
"${qual_scp[@]}" "$qual_repo/tests/agents/test_ge04_cgroup_worker.py" "$qual_target:$qual_remote/test_ge04_cgroup_worker.py"
# This script runs as ubuntu; each privileged call uses a fixed trusted program.
"${qual_ssh[@]}" "$qual_target" 'bash -s' > "$qual_artifacts/transcript.txt" 2>&1 <<'REMOTE'
set -euo pipefail
qual_root() { /usr/bin/sudo -n /usr/bin/env -i PATH=/usr/bin:/bin HOME=/nonexistent LANG=C "$@"; }
qual_source=/home/ubuntu/alix-ge04-workspaces/90c0042f-cgroup-01
qual_code=/run/ge04-cgroup-code
date -u +%Y-%m-%dT%H:%M:%SZ
# Refuse any preexisting stage. No adoption, recursive deletion or sudoers rule.
qual_root /usr/bin/gnumkdir -m 700 "$qual_code"
qual_root /usr/bin/gnumkdir -m 700 "$qual_code/native" "$qual_code/native/ge04" "$qual_code/tests" "$qual_code/tests/agents" "$qual_code/bin"
cat "$qual_source/cgroup-worker-qualification.c" | qual_root /usr/bin/tee "$qual_code/native/ge04/cgroup-worker-qualification.c" >/dev/null
cat "$qual_source/test_ge04_cgroup_worker.py" | qual_root /usr/bin/tee "$qual_code/tests/agents/test_ge04_cgroup_worker.py" >/dev/null
qual_root /usr/bin/gnuchmod 400 "$qual_code/native/ge04/cgroup-worker-qualification.c" "$qual_code/tests/agents/test_ge04_cgroup_worker.py"
qual_root /usr/bin/sha256sum --check <<'HASHES'
5362bdb0b08586f51151f0b53519a48ababb557f9c31758283f3aab2ec06d757  /run/ge04-cgroup-code/native/ge04/cgroup-worker-qualification.c
5f65166afc7a66c8d56370b40df32a55f16c82b4adb9fddfaa4eab05985d4174  /run/ge04-cgroup-code/tests/agents/test_ge04_cgroup_worker.py
HASHES
qual_root /usr/bin/cc -Wall -Wextra -Werror -O2 -o "$qual_code/bin/probe" "$qual_code/native/ge04/cgroup-worker-qualification.c"
qual_root /usr/bin/gnuchmod 500 "$qual_code/bin/probe"
# Fixed administrative manifest writer; no user/checkout/site imports.
qual_root /usr/bin/python3 -I -S - <<'MANIFEST'
import hashlib,json,os,pathlib
base=pathlib.Path('/run/ge04-cgroup-code')
paths={'sourceSha256':'native/ge04/cgroup-worker-qualification.c','controllerSha256':'tests/agents/test_ge04_cgroup_worker.py','binarySha256':'bin/probe'}
data={'schemaVersion':1,**{k:hashlib.sha256((base/p).read_bytes()).hexdigest() for k,p in paths.items()}}
assert data['sourceSha256']=='5362bdb0b08586f51151f0b53519a48ababb557f9c31758283f3aab2ec06d757' and data['controllerSha256']=='5f65166afc7a66c8d56370b40df32a55f16c82b4adb9fddfaa4eab05985d4174'
fd=os.open(base/'manifest.json',os.O_WRONLY|os.O_CREAT|os.O_EXCL|os.O_NOFOLLOW,0o400)
with os.fdopen(fd,'w') as out:json.dump(data,out,sort_keys=True);out.write('\n');out.flush();os.fsync(out.fileno())
fd=os.open(base,os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW);os.fsync(fd);os.close(fd)
print('STAGED_MANIFEST '+json.dumps(data,sort_keys=True))
MANIFEST
qual_stage_ids=$(
qual_root /usr/bin/python3 -I -S - <<'IDENTITIES'
import json,pathlib,stat
base=pathlib.Path('/run/ge04-cgroup-code')
files={'native/ge04/cgroup-worker-qualification.c':0o400,'tests/agents/test_ge04_cgroup_worker.py':0o400,'bin/probe':0o500,'manifest.json':0o400}
directories={'','native','native/ge04','tests','tests/agents','bin'}
observed={}
for path in [base,*base.rglob('*')]:
    name=str(path.relative_to(base)) if path!=base else ''
    m=path.lstat()
    assert (name in directories and stat.S_ISDIR(m.st_mode)) or (name in files and stat.S_ISREG(m.st_mode)),name
    assert m.st_uid==0 and stat.S_IMODE(m.st_mode)==(0o700 if name in directories else files[name])
    assert name in directories or m.st_nlink==1
    observed[name]=[m.st_dev,m.st_ino]
assert set(observed)==directories|set(files)
print(json.dumps({'schemaVersion':1,'entries':observed},sort_keys=True))
IDENTITIES
)
printf 'STAGING_IDENTITIES %s\n' "$qual_stage_ids"
printf 'COMMAND: /usr/bin/sudo -n /usr/bin/env -i PATH=/usr/bin:/bin HOME=/nonexistent LANG=C %q ALIX_GE04_PRIVILEGED_QUALIFICATION=1 ALIX_GE04_CGROUP_QUALIFICATION=1 /usr/bin/python3 -I -S /run/ge04-cgroup-code/tests/agents/test_ge04_cgroup_worker.py -v\n' "ALIX_GE04_STAGE_IDENTITIES=$qual_stage_ids"
set +e
qual_root ALIX_GE04_STAGE_IDENTITIES="$qual_stage_ids" ALIX_GE04_PRIVILEGED_QUALIFICATION=1 ALIX_GE04_CGROUP_QUALIFICATION=1 /usr/bin/python3 -I -S "$qual_code/tests/agents/test_ge04_cgroup_worker.py" -v
qual_result=$?
set -e
printf 'SUITE_EXIT=%s\n' "$qual_result"
# Preserve root staging for separately verified hash-matched cleanup/recovery.
exit "$qual_result"
REMOTE
# Persist the original bootstrap identity receipt separately from mutable remote state.
python3 - "$qual_artifacts" <<'SAVE_RECEIPT'
import json,pathlib,sys
base=pathlib.Path(sys.argv[1]);lines=(base/'transcript.txt').read_text().splitlines()
receipts=[line[len('STAGING_IDENTITIES '):] for line in lines if line.startswith('STAGING_IDENTITIES ')]
assert len(receipts)==1
value=json.loads(receipts[0]);assert set(value)=={'schemaVersion','entries'} and value['schemaVersion']==1
(base/'staging-identities.json').write_text(receipts[0]+'\n')
SAVE_RECEIPT
# No automatic teardown on failed/quarantined runs. Inspect the transcript/state.
printf '%s\n' "Transcript captured: $qual_artifacts/transcript.txt"
